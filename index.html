import asyncio from playwright.async_api import async_playwright # Список доменов трекеров, которые чекает бот TRACKER_DOMAINS = [ "mc.yandex.ru", # Яндекс.Метрика "top-fwz1.mail.ru", # VK / Top.Mail.ru "vk.com/r.php", # VK Пиксель "analytics.google.com", "googletagmanager.com" ] async def run_broken_test(url: str): async with async_playwright() as p: browser = await p.chromium.launch(headless=True) context = await browser.new_context() page = await context.new_page() # Наглухо блокируем запросы к пикселям — чекер их не увидит в сети async def block_trackers(route): request_url = route.request.url if any(domain in request_url for domain in TRACKER_DOMAINS): # Имитируем сбой сети или 500 ошибку трекера await route.abort("blockedbyclient") else: await route.continue_() await page.route("**/*", block_trackers) await page.goto(url, wait_until="networkidle") print("Страница загружена с заблокированными пикселями.") await browser.close() if __name__ == "__main__": asyncio.run(run_broken_test("http://localhost:3000"))
